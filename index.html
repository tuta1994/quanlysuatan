// 1. Lấy mã nhân viên vừa nhập từ Webhook const webhookData = $('Webhook').first().json; const maNhanVienMoi = String(webhookData.body ? webhookData.body.MaNhanVien : webhookData.MaNhanVien).trim(); // 2. Lấy toàn bộ danh sách từ node đọc Google Sheets ("DS đã đăng kí") const danhSachDangKy = $('DS đã đăng kí').all(); let tonTai = false; // 3. Duyệt qua danh sách và so sánh trực tiếp với cột 'Manhanviendangki' for (let item of danhSachDangKy) { const giaTriDangKy = item.json.Manhanviendangki || item.json.MaNhanViendadangki; if (giaTriDangKy !== undefined && giaTriDangKy !== null) { if (String(giaTriDangKy).trim() === maNhanVienMoi) { tonTai = true; break; // Trùng khớp thì dừng vòng lặp } } } // 4. Lấy thời gian hiện tại chuẩn định dạng Việt Nam const now = new Date(); const timeString = now.toLocaleTimeString('vi-VN', { timeZone: 'Asia/Ho_Chi_Minh' }); const dateString = now.toLocaleDateString('vi-VN', { timeZone: 'Asia/Ho_Chi_Minh' }); const thoiGianHienTai = `${timeString} ngày ${dateString}`; // 5. Trả về kết quả JSON chứa cả thông báo và thời gian để HTML dễ dàng xử lý if (tonTai) { return [{ json: { message: "Bạn đã đăng kí thành công", thoiGian: thoiGianHienTai } }]; } else { return [{ json: { message: "Bạn chưa đăng kí suất cơm", thoiGian: thoiGianHienTai } }]; }
